Option Explicit
Private Const cnsADDIN = "km_module.xla"   ' アドインファイル名
Private ExecuteFlag As Boolean
'--------------------------------------------------------
' セル変更によるイベント
'--------------------------------------------------------
Private Sub Worksheet_Change(ByVal Target As Range)
    If ExecuteFlag = False Then
        ExecuteFlag = True
        On Error Resume Next
        Workbooks(cnsADDIN).Application.Run ("CMD_WorkSheetExec"), arg1:=12, arg2:=Target.Address, arg3:=ThisWorkbook.Name, arg4:=ThisWorkbook.Path
        On Error GoTo 0
        ExecuteFlag = False
    End If
End Sub
'--------------------------------------------------------
' ダブルクリックによるイベント
'--------------------------------------------------------
Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    If ExecuteFlag = False Then
        ExecuteFlag = True
        On Error Resume Next
        Workbooks(cnsADDIN).Application.Run ("CMD_WorkSheetExec"), arg1:=15, arg2:=Target.Address, arg3:=ThisWorkbook.Name, arg4:=ThisWorkbook.Path
        On Error GoTo 0
        ExecuteFlag = False
    End If
End Sub
Private Function ButtonClickFunc2(no As Integer)
    ExecuteFlag = True
    On Error Resume Next
    Workbooks(cnsADDIN).Application.Run ("CMD_WorkSheetExec"), arg1:=no, arg2:=Null, arg3:=ThisWorkbook.Name, arg4:=ThisWorkbook.Path
    On Error GoTo 0
    ExecuteFlag = False
End Function
